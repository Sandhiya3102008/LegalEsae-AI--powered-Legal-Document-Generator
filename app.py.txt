from fastapi import FastAPI, UploadFile, File
from fastapi.middleware.cors import CORSMiddleware
import google.generativeai as genai

# Gemini API Key - inga unga key ah podu da
genai.configure(api_key="YOUR_API_KEY_HERE")

app = FastAPI()

# Frontend connect aaga
app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_methods=["*"],
    allow_headers=["*"],
)

@app.get("/")
def home():
    return {"message": "LegalEase Backend is Running!"}

@app.post("/analyze")
async def analyze_legal_doc(file: UploadFile = File(...)):
    text = (await file.read()).decode("utf-8", errors="ignore")
    
    model = genai.GenerativeModel("gemini-1.5-flash")
    prompt = f"Summarize this legal document in simple Tamil and English: \n\n{text[:5000]}"
    
    response = model.generate_content(prompt)
    return {"summary": response.text}
